if(FCC_WITH_FRANKA)
  if(FCC_FRANKA_SOURCE AND FCC_FRANKA_LIBRARY)
    # Also support audited wheel runtimes without installing into /usr/local.
    file(STRINGS "${FCC_FRANKA_SOURCE}/CMakeLists.txt" version_line REGEX "set\\(libfranka_VERSION [0-9.]+\\)")
    string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" Franka_VERSION "${version_line}")
    if(NOT Franka_VERSION OR NOT FCC_FRANKA_LIBRARY MATCHES "\\.so\\.${Franka_VERSION}$")
      message(FATAL_ERROR "libfranka header version must match the wheel shared library version")
    endif()
    add_library(Franka::Franka SHARED IMPORTED)
    set_target_properties(Franka::Franka PROPERTIES
      IMPORTED_LOCATION "${FCC_FRANKA_LIBRARY}"
      INTERFACE_INCLUDE_DIRECTORIES "${FCC_FRANKA_SOURCE}/include;${FCC_FRANKA_SOURCE}/common/include")
    get_filename_component(wheel_lib_dir "${FCC_FRANKA_LIBRARY}" DIRECTORY)
    # auditwheel hashes transitive dependency names; DT_RPATH applies to those too.
    target_link_options(_native PRIVATE "-Wl,--disable-new-dtags" "-Wl,-rpath,${wheel_lib_dir}")
  else()
    find_package(Franka CONFIG REQUIRED)
  endif()
  target_link_libraries(_native PRIVATE Franka::Franka)
  target_compile_definitions(_native PRIVATE FCC_WITH_FRANKA FCC_FRANKA_VERSION="${Franka_VERSION}")
  if(Franka_VERSION VERSION_GREATER_EQUAL "0.18")
    target_compile_definitions(_native PRIVATE FCC_OFFLINE_KINEMATICS)
  endif()
endif()
# Explicit opt-in offline bridge. Build only this target to leave a running
# hardware service's _native shared library untouched.
option(FCC_BUILD_TWIN "Build offline MuJoCo/controller bridge (libfranka >=0.18)" OFF)
if(FCC_BUILD_TWIN)
  if(NOT FCC_WITH_FRANKA OR Franka_VERSION VERSION_LESS "0.18")
    message(FATAL_ERROR "FCC_BUILD_TWIN requires libfranka >=0.18 for URDF-only dynamics")
  endif()
  file(SHA256 "${CMAKE_SOURCE_DIR}/native/controller.cpp" controller_source_sha256)
  set(twin_inputs "controller.cpp;controller.hpp;fr3_limits.hpp;operational_space.hpp;progressive_spring.hpp;joint_impedance.hpp;joint_limit_backoff.hpp;model_dynamics.hpp;twin_bindings.cpp")
  set(twin_fingerprints "")
  foreach(input IN LISTS twin_inputs)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "native/${input}")
    file(SHA256 "${CMAKE_SOURCE_DIR}/native/${input}" input_sha256)
    string(APPEND twin_fingerprints "${input}:${input_sha256}\n")
  endforeach()
  string(SHA256 twin_inputs_sha256 "${twin_fingerprints}")
  file(WRITE "${FCC_NATIVE_OUTPUT_DIRECTORY}/_native_inputs.json"
    "{\"twin_inputs_sha256\":\"${twin_inputs_sha256}\",\"controller_source_sha256\":\"${controller_source_sha256}\"}\n")
  pybind11_add_module(_twin_native native/twin_bindings.cpp native/controller.cpp)
  target_include_directories(_twin_native PRIVATE native)
  target_link_libraries(_twin_native PRIVATE Eigen3::Eigen ruckig Franka::Franka)
  target_compile_definitions(_twin_native PRIVATE
    FCC_FRANKA_VERSION="${Franka_VERSION}"
    FCC_CONTROLLER_SOURCE_SHA256="${controller_source_sha256}"
    FCC_TWIN_INPUTS_SHA256="${twin_inputs_sha256}")
  target_compile_options(_twin_native PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(_twin_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${FCC_NATIVE_OUTPUT_DIRECTORY}")
  if(wheel_lib_dir)
    target_link_options(_twin_native PRIVATE "-Wl,--disable-new-dtags" "-Wl,-rpath,${wheel_lib_dir}")
  endif()
endif()
